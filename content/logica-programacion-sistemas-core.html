Arquitectura General del Sistema

Todos los sistemas core de Total Darkness comparten un bus de eventos central (GameEventBus) y un sistema de estado global (GameStateManager) que orquesta la transición entre modos de juego: exploración, combate, QTE, cinemática y secuencias rítmicas.

Movimiento Híbrido

El GDD define una perspectiva híbrida: tercera persona limitada durante exploración y primera persona simbólica en rituales. El movimiento debe mezclar la fluidez arcade (DOOM) con la precisión táctica (Rainbow Six), adaptándose al contexto narrativo y al estado ético del jugador.

C# · Unity 
using UnityEngine;
using UnityEngine.InputSystem;

/// <summary>
/// Controlador de movimiento híbrido para Total Darkness.
/// Mezcla respuesta arcade (aceleración rápida) con física táctica (inercia, cobertura).
/// Se adapta al contexto: exploración, combate o ritual.
/// </summary>
public class CharacterMovementController : MonoBehaviour
{
    [Header("Referencias")]
    [SerializeField] private CharacterController controller;
    [SerializeField] private Transform cameraHolder;
    [SerializeField] private Animator animator;
    
    [Header("Curvas Arcade (respuesta rápida)")]
    [SerializeField] private float baseWalkSpeed = 4.5f;
    [SerializeField] private float sprintSpeed = 7.8f;
    [SerializeField] private float tacticalSpeed = 2.2f;
    [SerializeField] private float acceleration = 12f;   // Arcade: respuesta rápida
    [SerializeField] private float deceleration = 8f;
    
    [Header("Curvas Tácticas (inercia y peso)")]
    [SerializeField] private float momentumDecay = 0.92f; // Táctico: inercia
    [SerializeField] private float slopeLimit = 45f;
    [SerializeField] private float gravity = -20f;
    
    [Header("Modos de Perspectiva (GDD §4.5)")]
    [SerializeField] private float thirdPersonOffset = 2.5f;
    [SerializeField] private float firstPersonHeight = 1.7f;
    
    // Estado interno
    private Vector2 moveInput;
    private Vector3 velocity;
    private Vector3 verticalVelocity;
    private bool isSprinting, isAiming, isSliding;
    private MovementMode currentMode = MovementMode.Exploration;
    
    private enum MovementMode { Exploration, Combat, Ritual, Staggered }
    
    public void OnMove(InputAction.CallbackContext ctx) 
        => moveInput = ctx.ReadValue<Vector2>();
    
    public void OnSprint(InputAction.CallbackContext ctx) 
        => isSprinting = ctx.performed && currentMode == MovementMode.Exploration;
    
    public void OnAim(InputAction.CallbackContext ctx)
    {
        isAiming = ctx.performed;
        TransitionCamera(isAiming ? CameraMode.AimDown : CameraMode.ThirdPerson);
    }
    
    private void Update()
    {
        CalculateDesiredVelocity();
        ApplyGravity();
        ApplyMovement();
        UpdateAnimationParameters();
    }
    
    private void CalculateDesiredVelocity()
    {
        // Dirección relativa a la cámara (estándar shooter)
        Vector3 forward = cameraHolder.forward;
        Vector3 right = cameraHolder.right;
        forward.y = 0; right.y = 0;
        forward.Normalize(); right.Normalize();
        
        Vector3 desiredDir = (forward * moveInput.y + right * moveInput.x).normalized;
        
        // Velocidad objetivo según contexto
        float targetSpeed = currentMode switch
        {
            MovementMode.Exploration => isSprinting ? sprintSpeed : baseWalkSpeed,
            MovementMode.Combat => isAiming ? tacticalSpeed : baseWalkSpeed * 0.85f,
            MovementMode.Ritual => baseWalkSpeed * 0.6f, // Paso ceremonial lento
            MovementMode.Staggered => 0f,
            _ => baseWalkSpeed
        };
        
        // Mezcla arcade/táctica según el modo
        float blendFactor = currentMode == MovementMode.Combat 
            ? 0.5f  // Táctico: más peso, más inercia
            : 0.85f; // Arcade: respuesta rápida
        
        Vector3 targetVelocity = desiredDir * targetSpeed;
        velocity = Vector3.Lerp(velocity, targetVelocity, 
            (moveInput.sqrMagnitude > 0.01f ? acceleration : deceleration) 
            * blendFactor * Time.deltaTime);
    }
    
    private void ApplyGravity()
    {
        if (controller.isGrounded && verticalVelocity.y < 0)
            verticalVelocity.y = -2f;
        verticalVelocity.y += gravity * Time.deltaTime;
    }
    
    private void ApplyMovement()
    {
        Vector3 finalMove = velocity + verticalVelocity;
        controller.Move(finalMove * Time.deltaTime);
    }
    
    public void SetMode(MovementMode mode)
    {
        currentMode = mode;
        GameEventBus.Publish(new MovementModeChanged(mode));
    }
}

Transición de Perspectiva

Según el GDD (§4.5), el juego usa 3ª persona limitada en exploración y 1ª persona simbólica en rituales. La transición debe ser suave y diegética.

C# 
public void TransitionCamera(CameraMode target)
{
    StartCoroutine(SmoothTransition(target));
}

private IEnumerator SmoothTransition(CameraMode t)
{
    float elapsed = 0f;
    Vector3 startPos = cameraHolder.localPosition;
    Vector3 endPos = t == CameraMode.FirstPerson
        ? new Vector3(0, firstPersonHeight, 0)
        : new Vector3(0, 1.5f, -thirdPersonOffset);
    
    while (elapsed < 0.4f)
    {
        elapsed += Time.deltaTime;
        cameraHolder.localPosition = Vector3.Lerp(
            startPos, endPos, 
            EaseInOutCubic(elapsed / 0.4f));
        yield return null;
    }
}

Influencia del Estado Ético

El Eco Filosófico (GDD §2.3.4) modifica el movimiento: un jugador corrupto se mueve más pesado, uno redimido con mayor fluidez.

C# 
private void ApplyEthicalModifier()
{
    // -1.0 (corrupto) a +1.0 (redimido)
    float karma = GameStateManager.Instance.Karma;
    
    // Velocidad base afectada por karma
    float karmaModifier = 1f + (karma * 0.15f);
    baseWalkSpeed = 4.5f * karmaModifier;
    
    // Inercia: corrupto = más pesado
    momentumDecay = 0.92f - (karma * 0.05f);
    
    // Animación: blend hacia postura corrupta
    animator.SetFloat("Corruption", 
        Mathf.InverseLerp(-1, 1, -karma));
}

public class NarrativeState
{
    public float Karma { get; private set; } = 0f; // -1.0 a +1.0
    public float RhythmAccuracy { get; private set; } = 0.5f; // 0.0 a 1.0
    public int FragmentsCollected { get; private set; }
    public HashSet<string> ActiveFlags { get; } = new();
    public List<string> DecisionHistory { get; } = new();
    
    public void ApplyChoice(NarrativeChoice choice)
    {
        Karma = Mathf.Clamp(Karma + choice.karmaDelta, -1f, 1f);
        DecisionHistory.Add(choice.text);
        GameEventBus.Publish(new KarmaChanged(Karma));
    }
    
    public void UpdateRhythmAccuracy(float latestSync)
    {
        // Media móvil exponencial para suavizar
        RhythmAccuracy = RhythmAccuracy * 0.7f + latestSync * 0.3f;
    }
    
    public bool EvaluateCondition(Condition c) => c.type switch
    {
        ConditionType.Karma => Karma >= c.threshold,
        ConditionType.RhythmSync => RhythmAccuracy >= c.threshold,
        ConditionType.FragmentsCollected => FragmentsCollected >= (int)c.threshold,
        ConditionType.FlagSet => ActiveFlags.Contains(c.flagId),
        _ => false
    };
}

public class EndingEvaluator
{
    public EndingDefinition Evaluate(NarrativeState state, NarrativeGraph graph)
    {
        // 1. Prioridad: sincronía musical (GDD §1.4.2)
        if (state.RhythmAccuracy > 0.80f)
            return graph.endings.Find(e => e.id == "ARCHETYPAL_REDEMPTION");
        
        if (state.RhythmAccuracy >= 0.40f)
            return graph.endings.Find(e => e.id == "PERSISTENT_DUALITY");
        
        // 2. Si sincronía baja, decisión final del jugador
        string lastDecision = state.DecisionHistory[^1];
        return lastDecision.Contains("aceptar")
            ? graph.endings.Find(e => e.id == "LUCID_OBLIVION")
            : graph.endings.Find(e => e.id == "SYMBOL_REWRITE");
    }
}

Fórmula de Karma Compuesto

K_final = (0.5 × Σ decisiones) + (0.3 × sincronía) + (0.2 × exploración)

Las decisiones explícitas pesan 50%, la precisión rítmica 30%, y la exploración simbólica 20%. Esto garantiza que el jugador que solo dialogue sin explorar no alcance los finales más profundos.

  • METOR_KNOWS_TRUTH — desbloquea diálogo especial
  • REJECTED_ALL_PACTS — activa final alternativo
  • HEARD_PRIMORDIAL_SONG — modifica boss fight
  • SAKRIFICED_ECHO — desbloquea Redención

Quick Time Events

Los QTE en Total Darkness no son simples "presiona X". Son rituales sincronizados con la música de Redemthor y Mike Hellawaits. El sistema mezcla timing estricto (ventanas de 80ms) con ventanas tolerantes (300ms) según el contexto narrativo.

Ventana: 80ms · Penalización alta

Narrativo Tolerante

Ventana: 300ms · Penalización baja

Rítmico Musical

Sincronizado con BPM · Ventana variable

C# · QTE System 
using UnityEngine;
using System;
using System.Collections.Generic;

/// <summary>
/// Controlador de Quick Time Events con tres modos:
/// - Strict: ventana de 80ms, penalización alta
/// - Tolerant: ventana de 300ms, penalización baja  
/// - Rhythmic: sincronizado con BPM de la pista musical
/// </summary>
public class QTEController : MonoBehaviour
{
    [SerializeField] private AudioSource musicSource;
    [SerializeField] private CanvasGroup qteUI;
    
    private Queue<QTEPrompt> promptQueue = new();
    private QTEPrompt activePrompt;
    private float currentWindowStart, currentWindowEnd;
    private bool isActive;
    
    public event Action<QTEResult> OnPromptResolved;
    
    public void StartSequence(QTESequence sequence)
    {
        isActive = true;
        foreach (var p in sequence.prompts)
            promptQueue.Enqueue(p);
        SpawnNextPrompt();
    }
    
    private void SpawnNextPrompt()
    {
        if (promptQueue.Count == 0) { EndSequence(); return; }
        
        activePrompt = promptQueue.Dequeue();
        
        // Calcular ventana según el modo
        float windowSize = activePrompt.mode switch
        {
            QTEMode.Strict => 0.08f,    // 80ms
            QTEMode.Tolerant => 0.30f,  // 300ms
            QTEMode.Rhythmic => CalculateRhythmicWindow(),
            _ => 0.20f
        };
        
        // En modo rítmico, alinear ventana con el beat
        if (activePrompt.mode == QTEMode.Rhythmic)
        {
            float beatInterval = 60f / activePrompt.bpm;
            float nextBeat = Mathf.Ceiling(
                (float)AudioSettings.dspTime / beatInterval) * beatInterval;
            currentWindowStart = nextBeat - windowSize / 2f;
            currentWindowEnd = nextBeat + windowSize / 2f;
        }
        else
        {
            currentWindowStart = Time.time + activePrompt.delay;
            currentWindowEnd = currentWindowStart + windowSize;
        }
        
        ShowUI(activePrompt);
    }
    
    private void Update()
    {
        if (!isActive || activePrompt == null) return;
        
        // Verificar input del jugador
        if (CheckInput(activePrompt.expectedInput))
        {
            float now = activePrompt.mode == QTEMode.Rhythmic
                ? (float)AudioSettings.dspTime
                : Time.time;
            
            if (now >= currentWindowStart && now <= currentWindowEnd)
            {
                // Calcular precisión (0 = perfecto, 1 = fuera)
                float center = (currentWindowStart + currentWindowEnd) / 2f;
                float accuracy = Mathf.Clamp01(
                    Mathf.Abs(now - center) / (windowSize / 2f));
                
                QTEResult result = accuracy < 0.2f ? QTEResult.Perfect
                    : accuracy < 0.5f ? QTEResult.Good
                    : accuracy < 0.8f ? QTEResult.Ok
                    : QTEResult.Miss;
                
                OnPromptResolved?.Invoke(result);
                SpawnNextPrompt();
            }
        }
        
        // Timeout
        if (Time.time > currentWindowEnd + 0.1f)
        {
            OnPromptResolved?.Invoke(QTEResult.Fail);
            if (activePrompt.mode == QTEMode.Strict)
                HandleFailure(); // Penalización alta
            else
                SpawnNextPrompt(); // Continuar en tolerante
        }
    }
    
    private float CalculateRhythmicWindow()
    {
        // Ventana más generosa en BPM bajos, más estricta en altos
        float bpm = activePrompt.bpm;
        return Mathf.Lerp(0.25f, 0.08f, (bpm - 80f) / 120f);
    }
}

Integración QTE → Narrativa

C# 
// Cada QTE resuelto alimenta el NarrativeState
public class QTEToNarrativeBridge : MonoBehaviour
{
    [SerializeField] private QTEController qte;
    [SerializeField] private NarrativeState narrative;
    
    private void OnEnable() => qte.OnPromptResolved += HandleResult;
    private void OnDisable() => qte.OnPromptResolved -= HandleResult;
    
    private void HandleResult(QTEResult result)
    {
        float syncValue = result switch
        {
            QTEResult.Perfect => 1.0f,
            QTEResult.Good => 0.75f,
            QTEResult.Ok => 0.5f,
            QTEResult.Miss => 0.2f,
            QTEResult.Fail => 0.0f,
            _ => 0f
        };
        narrative.UpdateRhythmAccuracy(syncValue);
    }
}

Combate y Fatality

Inspirado en Killer Instinct y Mortal Kombat, el sistema de combate de Total Darkness usa secuencias de botones rítmicas para ejecutar ataques especiales y Fatalities. Las secuencias se validan con un input buffer y se sincronizan con el BPM de la música de fondo.

Input Buffer con Ventana Temporal

El buffer almacena las últimas entradas del jugador con timestamps. Cada frame se limpia lo obsoleto y se busca coincidencia con las secuencias definidas en los MoveDefinition.

C# · Input Buffer 
using System.Collections.Generic;
using UnityEngine;

public class InputBuffer
{
    private readonly List<BufferedInput> buffer = new();
    private readonly float maxAge = 0.6f; // 600ms de memoria
    
    public struct BufferedInput
    {
        public CombatInput input;
        public float timestamp;
        public float beatPhase; // Fase del beat al presionar (0-1)
    }
    
    public void Record(CombatInput input, float beatPhase)
    {
        buffer.Add(new BufferedInput 
        { 
            input = input, 
            timestamp = Time.time,
            beatPhase = beatPhase
        });
        Prune();
    }
    
    private void Prune()
    {
        float cutoff = Time.time - maxAge;
        buffer.RemoveAll(b => b.timestamp < cutoff);
    }
    
    /// <summary>
    /// Busca coincidencia con una secuencia definida.
    /// Tolerancia: cada input debe caer dentro de ±tolerance segundos
    /// del momento esperado.
    /// </summary>
    public MatchResult TryMatch(MoveDefinition move, float tolerance)
    {
        if (buffer.Count < move.sequence.Length) 
            return new MatchResult { matched = false };
        
        // Comparar desde el final del buffer hacia atrás
        int bufferIdx = buffer.Count - 1;
        int seqIdx = move.sequence.Length - 1;
        float totalRhythmScore = 0f;
        int rhythmHits = 0;
        
        while (seqIdx >= 0 && bufferIdx >= 0)
        {
            var expected = move.sequence[seqIdx];
            var actual = buffer[bufferIdx];
            
            if (actual.input == expected)
            {
                // Evaluar sincronía rítmica
                float phaseDiff = Mathf.Abs(actual.beatPhase - expected.beatTarget);
                phaseDiff = Mathf.Min(phaseDiff, 1f - phaseDiff); // wrap-around
                float rhythmQuality = 1f - Mathf.Clamp01(phaseDiff / 0.15f);
                totalRhythmScore += rhythmQuality;
                rhythmHits++;
                
                seqIdx--;
            }
            bufferIdx--;
        }
        
        bool matched = seqIdx < 0;
        float avgRhythm = rhythmHits > 0 ? totalRhythmScore / rhythmHits : 0f;
        
        return new MatchResult 
        { 
            matched = matched, 
            rhythmScore = avgRhythm 
        };
    }
}

C# · Combat Controller 
public class CombatController : MonoBehaviour
{
    [SerializeField] private InputBuffer buffer;
    [SerializeField] private List<MoveDefinition> availableMoves;
    [SerializeField] private RhythmTracker rhythmTracker;
    [SerializeField] private NarrativeState narrative;
    
    private Animator animator;
    private bool canAct = true;
    
    public void RecordInput(CombatInput input)
    {
        float beatPhase = rhythmTracker.GetCurrentPhase();
        buffer.Record(input, beatPhase);
        TryExecuteMove();
    }
    
    private void TryExecuteMove()
    {
        if (!canAct) return;
        
        // Priorizar Fatalities sobre combos
        foreach (var move in availableMoves)
        {
            if (move.isFatality && !CanExecuteFatality()) continue;
            
            var match = buffer.TryMatch(move, move.rhythmTolerance);
            if (match.matched)
            {
                ExecuteMove(move, match.rhythmScore);
                return;
            }
        }
    }
    
    private void ExecuteMove(MoveDefinition move, float rhythmScore)
    {
        canAct = false;
        
        // Daño escalado por precisión rítmica
        float finalDamage = move.damage * (0.7f + rhythmScore * 0.6f);
        
        // Ejecutar animación
        animator.Play(move.animation.name);
        
        // Spawn VFX
        Instantiate(move.vfxPrefab, transform.position, Quaternion.identity);
        
        // Impacto en narrativa (GDD §2.1.4)
        float karmaDelta = move.isFatality ? -0.05f : 0f;
        narrative.Karma += karmaDelta;
        
        // Restaurar capacidad de acción tras animación
        Invoke(nameof(RestoreAction), move.animation.length);
        
        GameEventBus.Publish(new MoveExecuted(move, rhythmScore));
    }
    
    private bool CanExecuteFatality()
    {
        var target = GetMeleeTarget();
        return target != null 
            && target.CurrentHP < target.MaxHP * 0.15f
            && target.IsStaggered;
    }
}

Sincronización Rítmica

La música de Redemthor y Mike Hellawaits no es solo ambientación: es el reloj maestro del juego. El RhythmTracker expone la fase del beat actual a todos los sistemas (QTE, combate, narrativa).

C# · Rhythm Tracker 
using UnityEngine;

/// <summary>
/// Rastrea el beat de la pista musical activa usando AudioSettings.dspTime
/// para máxima precisión (independiente de Time.time).
/// </summary>
public class RhythmTracker : MonoBehaviour
{
    [SerializeField] private AudioSource musicSource;
    [SerializeField] private float bpm = 120f;
    [SerializeField] private int beatsPerBar = 4;
    [SerializeField] private double songOffset = 0.0; // Ajuste manual de sync
    
    public float BeatInterval => 60f / bpm;
    public float BarInterval => BeatInterval * beatsPerBar;
    
    public event System.Action<int> OnBeat;
    public event System.Action<int> OnBar;
    
    private int lastBeat = -1;
    private int lastBar = -1;
    
    public float GetCurrentPhase()
    {
        double songTime = AudioSettings.dspTime - songOffset;
        double beatPos = songTime / BeatInterval;
        return (float)(beatPos - Mathf.Floor((float)beatPos));
    }
    
    public int GetCurrentBeat()
    {
        double songTime = AudioSettings.dspTime - songOffset;
        return (int)(songTime / BeatInterval);
    }
    
    private void Update()
    {
        int currentBeat = GetCurrentBeat();
        if (currentBeat != lastBeat)
        {
            lastBeat = currentBeat;
            OnBeat?.Invoke(currentBeat % beatsPerBar);
            
            int currentBar = currentBeat / beatsPerBar;
            if (currentBar != lastBar)
            {
                lastBar = currentBar;
                OnBar?.Invoke(currentBar);
            }
        }
    }
    
    /// <summary>
    /// Calcula qué tan cerca está el jugador del beat perfecto.
    /// 0 = perfecto, 1 = completamente fuera.
    /// </summary>
    public float EvaluateTiming(float inputTime)
    {
        float phase = GetCurrentPhase();
        float distance = Mathf.Min(phase, 1f - phase);
        return distance / 0.5f; // Normalizar a 0-1
    }
}

Integración Final

Todos los sistemas se orquestan desde el GameLoopManager, que cambia entre estados globales y delega a los sistemas especializados.

C# · Game Loop 
public class GameLoopManager : MonoBehaviour
{
    public enum GameState { Exploration, Combat, QTE, Ritual, Cinematic, Dialog }
    
    [SerializeField] private CharacterMovementController movement;
    [SerializeField] private CombatController combat;
    [SerializeField] private QTEController qte;
    [SerializeField] private NarrativeState narrative;
    [SerializeField] private RhythmTracker rhythm;
    
    private GameState currentState = GameState.Exploration;
    
    public void TransitionTo(GameState newState)
    {
        // Desactivar sistema anterior
        switch (currentState)
        {
            case GameState.Exploration: movement.SetMode(MovementMode.Exploration); break;
            case GameState.Combat: combat.Disable(); break;
            case GameState.QTE: qte.Stop(); break;
        }
        
        // Activar nuevo sistema
        switch (newState)
        {
            case GameState.Exploration: movement.SetMode(MovementMode.Exploration); break;
            case GameState.Combat: movement.SetMode(MovementMode.Combat); combat.Enable(); break;
            case GameState.QTE: movement.SetMode(MovementMode.Staggered); qte.Start(); break;
            case GameState.Ritual: movement.SetMode(MovementMode.Ritual); break;
        }
        
        currentState = newState;
        GameEventBus.Publish(new GameStateChanged(newState));
    }
    
    // Ejemplo de flujo: exploración → combate → fatality → QTE ritual
    public void OnEnemyDefeated(Enemy enemy)
    {
        if (enemy.WasKilledByFatality)
        {
            // Lanzar QTE ritual post-Fatality
            TransitionTo(GameState.QTE);
            qte.StartSequence(GetRitualSequence(enemy));
        }
        else
        {
            TransitionTo(GameState.Exploration);
        }
    }
}
