Ejemplo 1: Listar objetos en Blender

blender_list_objects.py · Blender bpy 4.x 
# Script para listar todos los objetos mesh en la escena activa
import bpy

def listar_meshes():
    """Retorna una lista de tuplas (nombre, polígonos, vértices)"""
    resultados = []
    for obj in bpy.context.scene.objects:
        if obj.type == 'MESH':
            mesh = obj.data
            resultados.append({
                'nombre': obj.name,
                'vertices': len(mesh.vertices),
                'caras': len(mesh.polygons),
                'tris': sum(len(p.vertices) - 2 for p in mesh.polygons)
            })
    return resultados

# Ejecutar y mostrar resultados
datos = listar_meshes()
for d in datos:
    print(f"{d['nombre']} → {d['tris']} tris")

Photoshop UXP: El Futuro de la Extensibilidad

La Unified Extensibility Platform (UXP) reemplaza a ExtendScript (ES3) con un motor JavaScript V8 moderno. Ofrece mejor rendimiento, estabilidad y acceso a APIs actualizadas. Todo nuevo desarrollo debe usar UXP.

❌ EXTENDSCRIPT (LEGADO)
JavaScript ES3, lento, sin soporte moderno. Aún funciona pero en desuso.
✓ UXP (MODERNO)
JavaScript V8, rápido, estable. Recomendado para todo desarrollo nuevo.

// Empaqueta Metallic (R), Roughness (G), AO (B) en un solo archivo ORM
const { app, action, core } = require('photoshop');
const fs = require('uxp').storage.localFileSystem;

async function packORM(metallicPath, roughnessPath, aoPath, outputPath) {
    await core.executeAsModal(async (ctx) => {
        // Abrir los tres mapas
        const metallic = await app.open(await fs.getFileForOpening());
        const roughness = await app.open(await fs.getFileForOpening());
        const ao = await app.open(await fs.getFileForOpening());
        
        // Crear documento RGB nuevo
        const doc = await app.createDocument({
            width: metallic.width,
            height: metallic.height,
            mode: 'RGB'
        });
        
        // Copiar canales a R, G, B
        await action.batchPlay([
            { _obj: "set", _target: [{_ref:"channel", _enum:"channel", _value:"red"}] },
            { _obj: "paste", _special: "into" }
        ], {});
        
        // Guardar como PNG optimizado
        await doc.save(await fs.getFileForSaving("ORM_output.png"));
    });
}

# Script para crear blockout humanoide básico
def crear_blockout():
    partes = {
        "torso": (0.4, 0.3, 0.6),
        "cabeza": (0.22, 0.22, 0.25),
        "brazo": (0.1, 0.1, 0.55),
        "pierna": (0.13, 0.13, 0.85)
    }
    for nombre, dims in partes.items():
        bpy.ops.mesh.primitive_cube_add(size=1)
        obj = bpy.context.active_object
        obj.name = nombre
        obj.scale = dims
        bpy.ops.object.transform_apply(scale=True)

CAPÍTULO III

Retopología y Optimización

La fase más crítica técnicamente. Convertir millones de polígonos en miles, manteniendo una topología limpia para animación y rendimiento.

Principios de Topología AAA

◈

Cuadriláteros

Preferir quads sobre tris y n-gons. Los quads subdividen limpiamente y deforman mejor.

◉

Flujo Anatómico

Los edge loops deben seguir las líneas musculares y las articulaciones (codos, rodillas, ojos, boca).

◇

Polos en Zonas Planas

Colocar vértices con 3 o 5+ conexiones en áreas de baja deformación (espalda, laterales).

Herramientas por Software

RetopoFlow + Quadriflow

RetopoFlow permite retopología manual precisa sobre el high-poly. Quadriflow (integrado) ofrece retopología automática de alta calidad basada en Instant Meshes.

quadriflow_script.py · Blender
import bpy

# Aplicar Quadriflow Remesher con parámetros óptimos
obj = bpy.context.active_object
bpy.ops.object.select_all(action='DESELECT')
obj.select_set(True)

# Configurar modificadores de optimización
decimate = obj.modifiers.new(name="Decimate", type='DECIMATE')
decimate.ratio = 0.1  # Reducir al 10%
decimate.use_collapse = True

# Aplicar Quadriflow para topología limpia
bpy.ops.object.quadriflow_remesh(
    use_mesh_bound=True,
    use_face_corners=False,
    res=10000  # Número objetivo de caras
)

print(f"Retopología completada: {len(obj.data.polygons)} caras")

⚠ Problemas Comunes y Soluciones

Triángulos visibles tras subdivisión

Solución: Convertir todos los tris a quads antes de exportar. En Blender: Mesh > Faces > Tris to Quads.

Pinchazos en articulaciones al animar

Solución: Añadir edge loops de soporte alrededor de codos/rodillas (3 loops mínimo). Usar pesos de vértices suaves.

Densidad de texel inconsistente

Solución: Mantener la misma densidad de texel en todas las islas UV. Usar UVPackmaster o herramientas nativas de empaquetado.

import bpy
from bpy.types import Operator, Panel

bl_info = {
    "name": "TD Topology Validator",
    "author": "Xlerion Studios",
    "version": (1, 0, 0),
    "blender": (4, 0, 0),
    "category": "Mesh"
}

class TD_OT_Validate(Operator):
    bl_idname = "td.validate_topology"
    bl_label = "Validar Topología"
    
    def execute(self, context):
        obj = context.active_object
        if not obj or obj.type != 'MESH':
            self.report({'WARNING'}, "Selecciona un mesh")
            return {'CANCELLED'}
        
        mesh = obj.data
        ngons = sum(1 for p in mesh.polygons if len(p.vertices) > 4)
        tris = sum(1 for p in mesh.polygons if len(p.vertices) == 3)
        quads = sum(1 for p in mesh.polygons if len(p.vertices) == 4)
        
        # Guardar resultados en propiedades de escena
        context.scene.td_ngons = ngons
        context.scene.td_tris = tris
        context.scene.td_quads = quads
        
        if ngons > 0:
            self.report({'WARNING'}, f"⚠ {ngons} n-gons detectados")
        else:
            self.report({'INFO'}, "✓ Topología limpia")
        
        return {'FINISHED'}

class TD_PT_Panel(Panel):
    bl_label = "TD Validator"
    bl_idname = "TD_PT_MainPanel"
    bl_space_type = 'VIEW_3D'
    bl_region_type = 'UI'
    bl_category = "Total Darkness"
    
    def draw(self, context):
        layout = self.layout
        scene = context.scene
        
        layout.operator("td.validate_topology", icon='CHECKMARK')
        
        col = layout.column(align=True)
        col.label(text=f"Quads: {scene.td_quads}")
        col.label(text=f"Tris: {scene.td_tris}")
        col.label(text=f"N-gons: {scene.td_ngons}")

def register():
    bpy.types.Scene.td_ngons = bpy.props.IntProperty(default=0)
    bpy.types.Scene.td_tris = bpy.props.IntProperty(default=0)
    bpy.types.Scene.td_quads = bpy.props.IntProperty(default=0)
    bpy.utils.register_class(TD_OT_Validate)
    bpy.utils.register_class(TD_PT_Panel)

def unregister():
    bpy.utils.unregister_class(TD_PT_Panel)

CAPÍTULO VI

Integración Inter-App

Cómo mover datos entre ZBrush, Blender, 3ds Max y Substance sin perder escala, rotación o información.

El Problema de GoZ/GoB

GoZ (ZBrush) y GoB (Blender) son puentes convenientes pero notoriamente frágiles. Los problemas más comunes:

  • ⚠ Escala incorrecta (modelo gigante o diminuto)
  • ⚠ Rotación errónea (ejes intercambiados)
  • ⚠ Posición bajo el suelo (origen mal configurado)
  • ⚠ Pérdida de UVs o grupos de polígonos

✓ Solución Profesional: Estándares Rígidos

1

Unidades Estandarizadas

Configurar TODAS las aplicaciones en centímetros. ZBrush: Preferences → Units. Blender: Scene Properties → Units. 3ds Max: Customize → Units Setup.

2

Punto de Origen Común

Crear un objeto de referencia (cubo de 200 cm) que sirva como ancla entre aplicaciones. Todos los personajes deben pivotar en (0,0,0).

3

Formatos Estándar

Usar FBX para personajes con rig, OBJ para esculturas estáticas, glTF/GLB para motores de juego modernos.

4

Scripts de Normalización

Scripts Python que al importar verifican y corrigen escala/rotación automáticamente.

# Normaliza escala y rotación al importar FBX en Blender
import bpy
import math

def importar_normalizado(ruta_fbx, escala_objetivo=1.0):
    # Importar con configuración fija
    bpy.ops.import_scene.fbx(
        filepath=ruta_fbx,
        use_manual_orientation=True,
        global_scale=escala_objetivo,
        axis_forward='-Z',
        axis_up='Y'
    )
    
    # Verificar y corregir escala
    for obj in bpy.context.selected_objects:
        if obj.type == 'MESH':
            # Aplicar transformaciones
            bpy.context.view_layer.objects.active = obj
            bpy.ops.object.transform_apply(
                location=True, 
                rotation=True, 
                scale=True
            )
            
            # Validar bounding box
            bbox = [obj.matrix_world @ math.Vector(corner) 
                    for corner in obj.bound_box]
            altura = max(v.z for v in bbox) - min(v.z for v in bbox)
            
            if altura > 300:  # > 3m = probablemente en mm
                obj.scale = (0.01, 0.01, 0.01)
                print("⚠ Escala corregida (mm → cm)")
    
    print("✓ Importación normalizada")

# Uso
importar_normalizado("C:/assets/character.fbx")

Pipeline AAA Estándar: High-Poly → Low-Poly

1

Concept Art (Photoshop)

Moodboards, character sheets, variaciones de estado ético.

2

Blockout (Maya/Blender)

Formas primarias con <5k tris. Validar silueta y proporciones.

3

Escultura (ZBrush)

Dynamesh → Subdividir → Esculpir detalles. 5-20M polígonos.

4

Retopología (Blender/Maya)

Crear low-poly con topología limpia. 8k-15k tris para personaje principal.

5

UV Mapping (RizomUV/Blender)

Densidad de texel uniforme, empaquetado eficiente, sin superposiciones.

6

Baking (Marmoset/Substance)

Normal, AO, Curvature, ID Maps del high-poly al low-poly.

7

Texturizado (Substance Painter)

PBR completo: Albedo, Normal, Metallic, Roughness, Emission.

8

Channel Packing (Photoshop)

Empaquetar ORM (Occlusion-Roughness-Metallic) en RGB.

9

Rigging (Maya/Blender)

Esqueleto, skinning, blend shapes para expresiones faciales.

10

Exportación a Unity/Unreal

FBX con configuración estandarizada. LODs, materiales PBR.
